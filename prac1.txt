BEGIN {
    rec = 1;
    send = 1;
    rate = 0;
}
{
    if ($1 == "+" && $5 == "tcp" && $3 == "0") {
        send = send + $6;
    }
    if ($1 == "r" && $5 == "tcp" && $4 == "3") {
        rec = rec + $6;
    }
}
END {
    rate = (rec / send) * 100;
    printf("\nPackets Sent: %d", send);
    printf("\nPackets Received: %d", rec);
    printf("\nSuccess Rate: %.2f%%\n", rate);
    printf("%.2f %.2f\n", (send - 1) / 1000000.0, rate) >> "result1.log";
}
