# sw_analyze.awk
# Usage: awk -v src=0 -v dst=1 -v rtt=0.4 -v bw=5 -v csv=1 -f sw_analyze.awk tracefile

BEGIN {
    src = (src == "" ? 0 : src)
    dst = (dst == "" ? 1 : dst)
    rtt = (rtt == "" ? 0.4 : rtt)
    bw_mbps = (bw == "" ? 0.2 : bw)   # give bw in Mbps for calculation only
    csv = (csv == "" ? 0 : csv)

    sent_pkts = 0
    sent_bytes = 0
    recv_pkts = 0
    recv_bytes = 0
}

{
    event = $1
    proto = $5
    from = $3 + 0
    to   = $4 + 0
    size = $6 + 0

    if (event == "+" && proto == "tcp" && from == src) {
        sent_pkts++
        sent_bytes += size
    }
    if (event == "r" && proto == "tcp" && to == dst) {
        recv_pkts++
        recv_bytes += size
    }
}

END {
    data_bits = sent_bytes * 8
    bw_bps = bw_mbps * 1000000.0
    bdp = rtt * bw_bps
    link_util = (bdp > 0) ? (data_bits / bdp) : 0

    if (csv == 1) {
        printf("%.3f,%d,%.6f,%.0f,%.6f\n", bw_mbps, data_bits, rtt, bdp, link_util)
    } else {
        printf("BW_Mbps: %.3f\nData_bits: %d\nRTT_s: %.6f\nBDP_bits: %.0f\nLink_util: %.6f\n", bw_mbps, data_bits, rtt, bdp, link_util)
    }
}






#!/bin/bash
# run_using_original_tcl.sh
# Place this in same folder as A1-stop-n-wait.tcl
# Requires: ns, awk, sed, bc

ORIG_TCL="A1-stop-n-wait.tcl"
OUT_CSV="sw_results.csv"

if [ ! -f "$ORIG_TCL" ]; then
  echo "ERROR: $ORIG_TCL not found in current directory."
  exit 1
fi

# RTT: your TCL uses 200ms propagation delay => RTT = 0.4 s
RTT=0.4

# header
echo "bw_mbps,data_bits,rtt_s,bdp_bits,link_utilization" > $OUT_CSV

# bandwidth sweep: 5.0 down to 1.0 in steps of 0.5
bw=5.0
while (( $(echo "$bw >= 1.0" | bc -l) )); do
  bwstr="${bw}Mb"
  echo "=== Running with bandwidth = $bwstr ==="

  # create temp tcl with bandwidth replaced (replace first occurrence of "0.2Mb")
  TMP_TCL="tmp_stopwait_${bw//./_}.tcl"
  # Use sed to replace first occurrence only
  sed "0,/0.2Mb/{s/0.2Mb/${bw}Mb/}" "$ORIG_TCL" > "$TMP_TCL"

  # run NS2
  ns "$TMP_TCL"

  # expected NS output trace file
  TR="A1-stop-n-wait.tr"
  if [ ! -f "$TR" ]; then
    echo "ERROR: trace $TR not produced. Check NS run output. Aborting."
    rm -f "$TMP_TCL"
    exit 1
  fi

  # rename trace to keep it
  TR_OUT="A1-stop-n-wait_${bw//./_}.tr"
  mv "$TR" "$TR_OUT"

  # analyze and append csv
  awk -v src=0 -v dst=1 -v rtt=$RTT -v bw=$bw -v csv=1 -f sw_analyze.awk "$TR_OUT" >> $OUT_CSV

  # cleanup temp tcl
  rm -f "$TMP_TCL"

  # decrement
  bw=$(echo "$bw - 0.5" | bc)
done

echo "All runs done. Results -> $OUT_CSV"
cat $OUT_CSV
